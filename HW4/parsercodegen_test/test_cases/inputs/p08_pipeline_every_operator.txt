/* every operator, each result written */
const k := 7;
var a, b;
begin
  a = 17;
  b = -5;
  write a + b;
  write a - b;
  write a * b;
  write a / b;
  write -a / k;
  if a == 17 then write 1 fi;
  if a != b then write 2 fi;
  if b < a then write 3 fi;
  if a <= 17 then write 4 fi;
  if a > b then write 5 fi;
  if b >= -5 then write 6 fi
end.
